/*
============================================================
  โจทย์ข้อที่ 1 : หาผลรวมและค่าเฉลี่ย
============================================================

  [โจทย์]
  จงเขียนโปรแกรมรับจำนวนเต็ม N ตัวจากผู้ใช้
  แล้วคำนวณและแสดงผล :
    1) ผลรวมของตัวเลขทั้งหมด
    2) ค่าเฉลี่ย (ทศนิยม 2 ตำแหน่ง)
    3) จำนวนตัวที่เป็นเลขคู่
    4) จำนวนตัวที่เป็นเลขคี่

  ตัวอย่าง Input :
    N = 5
    ใส่ตัวเลข : 3 8 5 2 7

  ตัวอย่าง Output :
    ผลรวม    = 25
    ค่าเฉลี่ย = 5.00
    เลขคู่   = 2 ตัว  (8, 2)
    เลขคี่   = 3 ตัว  (3, 5, 7)

============================================================
  [วิธีคิด]
  1. ใช้ for loop รับค่าเก็บใน array
  2. ใช้ for loop วนหาผลรวม + นับคู่/คี่
     - ถ้า arr[i] % 2 == 0 → เลขคู่
     - ถ้า arr[i] % 2 != 0 → เลขคี่
  3. ค่าเฉลี่ย = (float)sum / n  (cast เป็น float ก่อน หาร)
  4. printf("%.2f") สำหรับทศนิยม 2 ตำแหน่ง
============================================================
*/

#include <cstdio>

// --- function คำนวณผลรวม ---
int calcSum(int arr[], int n) {         // รับ array และขนาด n เข้ามา
    int sum = 0;                        // ตัวแปรเก็บผลรวม เริ่มต้นที่ 0
    for (int i = 0; i < n; i++) {       // วน loop ตั้งแต่ index 0 ถึง n-1
        sum += arr[i];                  // บวกค่าแต่ละตัวเข้าสะสมใน sum (sum = sum + arr[i])
    }
    return sum;                         // คืนค่าผลรวมกลับไป
}

// --- function นับเลขคู่ ---
int countEven(int arr[], int n) {       // รับ array และขนาด n เข้ามา
    int count = 0;                      // ตัวนับเลขคู่ เริ่มต้นที่ 0
    for (int i = 0; i < n; i++) {       // วน loop ตรวจทุกตัวใน array
        if (arr[i] % 2 == 0) count++;   // ถ้าหารด้วย 2 เหลือเศษ 0 = เลขคู่ → เพิ่ม count
    }
    return count;                       // คืนจำนวนเลขคู่ที่นับได้
}

// --- function แสดงเฉพาะเลขคู่หรือคี่ ---
//     type = 0 → แสดงเลขคู่ , type = 1 → แสดงเลขคี่
void printEvenOdd(int arr[], int n, int type) {
    printf("(");
    int first = 1;                      // flag บอกว่าเป็นตัวแรกที่พิมพ์หรือไม่ (1=ใช่, 0=ไม่ใช่)
    for (int i = 0; i < n; i++) {       // วน loop ทุกตัวใน array
        if (arr[i] % 2 == type) {       // เงื่อนไข : arr[i]%2==0 คือคู่, arr[i]%2==1 คือคี่
            if (!first) printf(", ");   // ถ้าไม่ใช่ตัวแรก ให้พิมพ์ ", " คั่นก่อน (!first = first เป็น 0)
            printf("%d", arr[i]);       // พิมพ์ตัวเลข
            first = 0;                  // เปลี่ยน flag เป็น 0 หมายถึงพิมพ์ตัวแรกไปแล้ว
        }
    }
    printf(")");
}

int main() {
    printf("========================================\n");
    printf("  โจทย์ 1 : ผลรวม ค่าเฉลี่ย เลขคู่-คี่\n");
    printf("========================================\n\n");

    int n;                              // ตัวแปรเก็บจำนวนตัวเลขที่จะรับ
    printf("ใส่จำนวนตัวเลข N : ");
    scanf("%d", &n);                    // รับค่า n จากผู้ใช้

    int arr[100];                       // ประกาศ array ขนาด 100 ช่อง (รองรับ n ได้สูงสุด 100 ตัว)
    printf("ใส่ตัวเลข %d ตัว : ", n);
    for (int i = 0; i < n; i++) {       // วน loop รับค่าทีละตัว
        scanf("%d", &arr[i]);           // รับค่าและเก็บใน arr[i] (ต้องใส่ & เพราะ scanf ต้องการ address)
    }

    // คำนวณ
    int sum     = calcSum(arr, n);      // เรียก function หาผลรวม → เก็บใน sum
    float avg   = (float)sum / n;       // (float) คือ type casting บังคับให้หารแบบทศนิยม (ไม่ใช่ int หาร int)
    int evenCnt = countEven(arr, n);    // เรียก function นับเลขคู่ → เก็บใน evenCnt
    int oddCnt  = n - evenCnt;          // จำนวนเลขคี่ = จำนวนทั้งหมด - จำนวนเลขคู่

    // แสดงผล
    printf("\n--- ผลลัพธ์ ---\n");
    printf("ผลรวม    = %d\n", sum);
    printf("ค่าเฉลี่ย = %.2f\n", avg);  // %.2f = แสดงทศนิยม 2 ตำแหน่ง
    printf("เลขคู่   = %d ตัว  ", evenCnt);
    printEvenOdd(arr, n, 0);            // ส่ง 0 = ขอแสดงเลขคู่
    printf("\n");
    printf("เลขคี่   = %d ตัว  ", oddCnt);
    printEvenOdd(arr, n, 1);            // ส่ง 1 = ขอแสดงเลขคี่
    printf("\n");

    return 0;                           // คืนค่า 0 = โปรแกรมจบปกติ
}
